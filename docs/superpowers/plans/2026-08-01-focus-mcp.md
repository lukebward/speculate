/** Remove artifacts a ≤0.10 install left behind (plugin, workspace server). */
export async function cleanupLegacyArtifacts(ctx: Ctx, view: HostView): Promise<void> {
  if (effectiveServers(view.servers).has(WORKSPACE_SERVER_NAME)) {
    const res = await ctx.runner(ctx.claudeBin,
      ['mcp', 'remove', WORKSPACE_SERVER_NAME, '-s', 'local'], { cwd: ctx.cwd });
    ctx.log(res.ok
      ? `[speculate] legacy: removed ${WORKSPACE_SERVER_NAME} (CLI speculation was retired in 0.11)`
      : `[speculate] legacy: could not remove ${WORKSPACE_SERVER_NAME}: ${(res.stderr || res.stdout).trim()}`);
  }
  const list = await ctx.runner(ctx.claudeBin, ['plugin', 'list', '--json'], { cwd: ctx.cwd });
  if (list.ok && list.stdout.includes('"speculate"')) {
    const un = await ctx.runner(ctx.claudeBin,
      ['plugin', 'uninstall', '-s', 'local', 'speculate'], { cwd: ctx.cwd });
    ctx.log(un.ok
      ? '[speculate] legacy: uninstalled the speculate plugin (Bash hook retired in 0.11)'
      : `[speculate] legacy: plugin uninstall failed — remove manually: ${ctx.claudeBin} plugin uninstall -s local speculate`);
    if (un.ok) await ctx.runner(ctx.claudeBin, ['plugin', 'marketplace', 'remove', 'speculate'], { cwd: ctx.cwd });
  }
}
```

## v0.11 (2026-08-01): MCP-only focus

CLI speculation (exec daemon, Bash hook, workspace shell server) is removed.
Rationale: speculation value scales with upstream latency — MCP/SaaS reads
(hundreds of ms) dominate local CLI reads (~30 ms, often net-negative after
hook spawn overhead); read-only vetting of arbitrary argv has no
deterministic cross-platform answer short of per-OS sandbox machinery,
while MCP's readOnlyHint gives it for free; and the tier carried the
project's POSIX-only surface (unix sockets, uid checks) plus a Windows
.git/index watcher loop that flushed every prefetch. `on`/`off` now clean
up artifacts a ≤0.10 install left behind. Full trail:
docs/superpowers/specs/2026-08-01-focus-mcp-design.md.
```

rmSync(dir, { recursive: true, force: true });
